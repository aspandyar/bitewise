# Этап 2 — база продуктов: план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** таблица продуктов на 100 г с поиском по 1–2 буквам и с опечатками, свои продукты и порции, копия общего продукта, сид ~200 продуктов из USDA и с сайтов производителей.

**Architecture:** Postgres (`foods`, `food_portions`, `pg_trgm`) → репозиторий `lib/foods/repo.ts` с фильтром видимости в каждом запросе → GET `/api/foods/search` (тонкий route, логика в чистом `handleFoodSearch`) и server actions → экраны `app/(app)/foods/*`. Сид собирается скриптом из CSV USDA + JSON-описаний в `db/seed/foods.json` и грузится upsert'ом по `seed_key`.

**Tech Stack:** Next.js 16.4 (App Router, `cacheComponents`), React 19, TypeScript strict, Drizzle ORM 0.45 + drizzle-kit 0.31, postgres.js, zod 4, Vitest 5, Tailwind v4.

**Spec:** `docs/superpowers/specs/2026-10-10-foods-design.md`

## Global Constraints
- Калории и БЖУ считает только `lib/nutrition.ts`; числа в сиде — только из данных USDA или со страницы производителя, не из памяти.
- Все числа и лимиты — в `lib/config.ts`; все тексты интерфейса — в `lib/i18n/ru.ts`.
- Комментарии, идентификаторы, сообщения логов и коммитов — на английском; коммиты — Conventional Commits, с трейлером `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Нутриенты — `numeric(8,2)`, `mode: "number"`; углеводы хранятся без клетчатки.
- Каждый запрос к `foods` / `food_portions` фильтрует видимость: продукт — `deleted_at IS NULL AND (owner_id IS NULL OR owner_id = U)`; порция — `user_id IS NULL OR user_id = U`. Изменять — только своё. Иначе 404.
- Каждый server action и route handler: сессия первой строкой → zod → репозиторий.
- Схема — только через `/db-change`: `pnpm db:generate` → показать SQL → «ок» пользователя → `pnpm db:migrate`. `db/migrations/**` не править (запрещено настройками), `drizzle-kit push` не использовать.
- Новых зависимостей нет.
- Mobile-first 375 px, тап-зоны ≥ 44 px (`min-h-11`), у каждого экрана — загрузка, пусто, ошибка.
- Push, merge — только после «ок» пользователя. Скачивание архива USDA — только после отдельного «ок».

```ts
function assertNonNegative(name: string, value: number): void {
  if (!(Number.isFinite(value) && value >= 0)) {
    throw new RangeError(`${name} must be a non-negative number, got ${value}`);
  }
}

// USDA reports carbs "by difference", which includes fiber; Russian labels and tables do not.
// Storing net carbs keeps seeded foods comparable with what users read on packages.
export function calcNetCarbsPer100g(carbsByDifferenceG: number, fiberG: number | null): number {
  assertNonNegative("carbsByDifferenceG", carbsByDifferenceG);
  if (fiberG === null) return carbsByDifferenceG;
  assertNonNegative("fiberG", fiberG);
  // Lab rounding can put fiber above carbs by difference; negative carbs are meaningless.
  return Math.max(0, carbsByDifferenceG - fiberG);
}
```

```ts
import { FOOD_SEARCH } from "@/lib/config";

// Mirrors foods.search_name (lower + ё→е) so the query and the column compare like for like.
export function normalizeFoodQuery(raw: string | null): string | null {
  if (raw === null) return null;
  const normalized = raw.trim().replace(/\s+/g, " ").toLowerCase().replaceAll("ё", "е");
  if (normalized.length === 0 || normalized.length > FOOD_SEARCH.maxQueryLength) return null;
  return normalized;
}

// Backslash is Postgres' default LIKE escape; without this "%" would match every food.
export function escapeLike(text: string): string {
  return text.replace(/[\\%_]/g, "\\$&");
}
```

```ts
import { FOOD_LIMITS, PORTION_LIMITS } from "@/lib/config";
import { decimalField, optionalDecimalField } from "@/lib/forms";
import { ru } from "@/lib/i18n/ru";

const { nameLength, kcal100, nutrient100G, maxTotalGramsPer100G } = FOOD_LIMITS;
const gramsMessage = ru.errors.gramsRange(nutrient100G.min, nutrient100G.max);
const grams = () => decimalField(nutrient100G, gramsMessage);
const nameMessage = ru.errors.nameLength(nameLength.min, nameLength.max);

// Integer hundredths: 33.3 + 33.3 + 33.4 is 100.00000000000001 in floats.
const toHundredths = (value: number) => Math.round(value * 100);

export const foodSchema = z
  .object({
    name: z
      .string()
      .trim()
      .min(nameLength.min, { error: nameMessage })
      .max(nameLength.max, { error: nameMessage }),
    kcal100: decimalField(kcal100, ru.errors.kcalRange(kcal100.min, kcal100.max)),
    protein100: grams(),
    fat100: grams(),
    carbs100: grams(),
    fiber100: optionalDecimalField(nutrient100G, gramsMessage),
  })
  .refine(
    (food) =>
      toHundredths(food.protein100) +
        toHundredths(food.fat100) +
        toHundredths(food.carbs100) +
        toHundredths(food.fiber100 ?? 0) <=
      toHundredths(maxTotalGramsPer100G),
    { error: ru.errors.macroSum(maxTotalGramsPer100G), path: ["carbs100"] },
  );

const { labelLength, grams: portionGrams } = PORTION_LIMITS;
const labelMessage = ru.errors.portionLabelLength(labelLength.min, labelLength.max);

export const portionSchema = z.object({
  label: z
    .string()
    .trim()
    .min(labelLength.min, { error: labelMessage })
    .max(labelLength.max, { error: labelMessage }),
  grams: decimalField(portionGrams, ru.errors.portionGrams(portionGrams.min, portionGrams.max)),
});

export const foodIdSchema = z.uuid();

// Ids arrive from URLs and bound action arguments; Postgres would throw on a malformed uuid.
export function parseFoodId(value: unknown): string | null {
  const parsed = foodIdSchema.safeParse(value);
  return parsed.success ? parsed.data : null;
```

```ts
import { and, asc, count, desc, eq, isNull, or, sql } from "drizzle-orm";
import { foodPortions, foods } from "@/db/schema";
import type { Db } from "@/db/types";
import { PORTION_LIMITS } from "@/lib/config";
import type {
  FoodDetails,
  FoodListItem,
  FoodValues,
  PortionItem,
  PortionValues,
} from "./types";

// Every read goes through this filter: shared or own, and never soft-deleted.
export function visibleTo(userId: string) {
  return and(isNull(foods.deletedAt), or(isNull(foods.ownerId), eq(foods.ownerId, userId)));
}

function ownedBy(userId: string, foodId: string) {
  return and(eq(foods.id, foodId), eq(foods.ownerId, userId), isNull(foods.deletedAt));
}

// A visible food with an owner can only be the caller's, so this never reveals other users.
const isOwnColumn = sql<boolean>`${foods.ownerId} IS NOT NULL`;

export const listColumns = {
  id: foods.id,
  name: foods.name,
  kcal100: foods.kcal100,
  protein100: foods.protein100,
  fat100: foods.fat100,
  carbs100: foods.carbs100,
  source: foods.source,
  isOwn: isOwnColumn,
};

export async function getVisibleFood(
  db: Db,
  userId: string,
  foodId: string,
): Promise<FoodDetails | null> {
  const [row] = await db
    .select({ ...listColumns, fiber100: foods.fiber100, sourceUrl: foods.sourceUrl })
    .from(foods)
    .where(and(eq(foods.id, foodId), visibleTo(userId)))
    .limit(1);
  return row ?? null;
}

export async function listOwnFoods(
  db: Db,
  userId: string,
  limit: number,
): Promise<FoodListItem[]> {
  return db
    .select(listColumns)
    .from(foods)
    .where(and(eq(foods.ownerId, userId), isNull(foods.deletedAt)))
    .orderBy(desc(foods.createdAt), desc(foods.id))
    .limit(limit);
}

export async function createFood(
  db: Db,
  userId: string,
  values: FoodValues,
): Promise<{ id: string }> {
  const [row] = await db
    .insert(foods)
    .values({ ...values, ownerId: userId, source: "user" })
    .returning({ id: foods.id });
  if (!row) throw new Error("food insert returned no row");
  return row;
}

export async function updateOwnFood(
  db: Db,
  userId: string,
  foodId: string,
  values: FoodValues,
): Promise<boolean> {
  const rows = await db
    .update(foods)
    .set({ ...values, updatedAt: sql`now()` })
    .where(ownedBy(userId, foodId))
    .returning({ id: foods.id });
  return rows.length > 0;
}

export async function softDeleteOwnFood(db: Db, userId: string, foodId: string): Promise<boolean> {
  const rows = await db
    .update(foods)
    .set({ deletedAt: sql`now()`, updatedAt: sql`now()` })
    .where(ownedBy(userId, foodId))
    .returning({ id: foods.id });
  return rows.length > 0;
}

export async function listVisiblePortions(
  db: Db,
  userId: string,
  foodId: string,
): Promise<PortionItem[]> {
  return db
    .select({
      id: foodPortions.id,
      label: foodPortions.label,
      grams: foodPortions.grams,
      isOwn: sql<boolean>`${foodPortions.userId} IS NOT NULL`,
    })
    .from(foodPortions)
    .innerJoin(foods, eq(foods.id, foodPortions.foodId))
    .where(
      and(
        eq(foodPortions.foodId, foodId),
        visibleTo(userId),
        or(isNull(foodPortions.userId), eq(foodPortions.userId, userId)),
      ),
    )
    // Shared portions first, then the user's own in the order they were added; copies insert
    // all portions in one statement, so the label breaks created_at ties deterministically.
    .orderBy(
      sql`${foodPortions.userId} IS NOT NULL`,
      asc(foodPortions.createdAt),
      asc(foodPortions.label),
    );
}

// Copies keep the numbers and every portion the user could see, so nothing is lost by copying.
export async function copyFood(
  db: Db,
  userId: string,
  foodId: string,
): Promise<{ id: string } | null> {
  return db.transaction(async (tx) => {
    const [source] = await tx
      .select({
        name: foods.name,
        kcal100: foods.kcal100,
        protein100: foods.protein100,
        fat100: foods.fat100,
        carbs100: foods.carbs100,
        fiber100: foods.fiber100,
      })
      .from(foods)
      .where(and(eq(foods.id, foodId), visibleTo(userId)))
      .limit(1);
    if (!source) return null;
    const copy = await createFood(tx, userId, source);
    const portions = await listVisiblePortions(tx, userId, foodId);
    if (portions.length > 0) {
      await tx.insert(foodPortions).values(
        portions.map(({ label, grams }) => ({ foodId: copy.id, userId, label, grams })),
      );
    }
    return copy;
  });
}

export async function addPortion(
  db: Db,
  userId: string,
  foodId: string,
  portion: PortionValues,
): Promise<"ok" | "not_found" | "limit"> {
  return db.transaction(async (tx) => {
    const [food] = await tx
      .select({ id: foods.id })
      .from(foods)
      .where(and(eq(foods.id, foodId), visibleTo(userId)))
      .limit(1);
    if (!food) return "not_found";
    // Two simultaneous submits can overshoot by one; the cap is a UX limit, not a security one.
    const [owned] = await tx
      .select({ total: count() })
      .from(foodPortions)
      .where(and(eq(foodPortions.foodId, foodId), eq(foodPortions.userId, userId)));
    if ((owned?.total ?? 0) >= PORTION_LIMITS.maxOwnPerFood) return "limit";
    await tx.insert(foodPortions).values({ foodId, userId, ...portion });
    return "ok";
  });
}

export async function deleteOwnPortion(
  db: Db,
  userId: string,
  portionId: string,
): Promise<boolean> {
  const rows = await db
    .delete(foodPortions)
    .where(and(eq(foodPortions.id, portionId), eq(foodPortions.userId, userId)))
    .returning({ id: foodPortions.id });
  return rows.length > 0;
}
```

```ts
// `query` must come from normalizeFoodQuery: it already matches search_name's lower/ё→е form.
export async function searchFoods(
  db: Db,
  userId: string,
  query: string,
): Promise<FoodListItem[]> {
  const escaped = escapeLike(query);
  const namePrefix = `${escaped}%`;
  const wordPrefix = `% ${escaped}%`;
  const startsWord = sql`(${foods.searchName} LIKE ${namePrefix} OR ${foods.searchName} LIKE ${wordPrefix})`;
  const match =
    query.length >= FOOD_SEARCH.minTrigramLength
      ? sql`(${startsWord} OR ${query} <% ${foods.searchName})`
      : startsWord;

  return db.transaction(async (tx) => {
    // `<%` reads its cut-off from this setting; LOCAL keeps it inside this transaction.
    await tx.execute(
      sql`SELECT set_config('pg_trgm.word_similarity_threshold', ${String(FOOD_SEARCH.wordSimilarityThreshold)}, true)`,
    );
    return tx
      .select(listColumns)
      .from(foods)
      .where(and(visibleTo(userId), match))
      .orderBy(
        sql`${foods.ownerId} IS NULL`,
        sql`${foods.searchName} LIKE ${namePrefix} DESC`,
        sql`${foods.searchName} LIKE ${wordPrefix} DESC`,
        sql`word_similarity(${query}, ${foods.searchName}) DESC`,
        asc(foods.name),
      )
      .limit(FOOD_SEARCH.limit);
  });
}
```

```ts
import type { RateLimiter } from "@/lib/auth/rate-limit";
import { normalizeFoodQuery } from "./search";
import type { FoodListItem } from "./types";

export type FoodSearchResponse =
  | { status: 200; body: { items: FoodListItem[] } }
  | {
      status: 400 | 401 | 429 | 500;
      body: { error: "bad_query" | "unauthorized" | "rate_limited" | "server_error" };
    };

// Kept free of Next so every branch is unit-testable; route.ts only wires request and response.
export async function handleFoodSearch(
  input: { userId: string | null; rawQuery: string | null; nowMs: number },
  deps: {
    limiter: RateLimiter;
    search: (userId: string, query: string) => Promise<FoodListItem[]>;
  },
): Promise<FoodSearchResponse> {
  if (!input.userId) return { status: 401, body: { error: "unauthorized" } };
  if (!deps.limiter.hit(input.userId, input.nowMs)) {
    return { status: 429, body: { error: "rate_limited" } };
  }
  const query = normalizeFoodQuery(input.rawQuery);
  if (!query) return { status: 400, body: { error: "bad_query" } };
  try {
    return { status: 200, body: { items: await deps.search(input.userId, query) } };
  } catch (error) {
    // Details stay in server logs; the client only learns that the search failed.
    console.error("food search failed", error);
    return { status: 500, body: { error: "server_error" } };
  }
}
```

```ts
import { NUTRIENT_STORAGE_DECIMALS, USDA_NUTRIENT_IDS } from "@/lib/config";
import { calcNetCarbsPer100g } from "@/lib/nutrition";
import type { LabelSource, SeedFood, UsdaSource } from "./seed-schema";

type NutrientKey = keyof typeof USDA_NUTRIENT_IDS;
export type UsdaNutrients = Partial<Record<NutrientKey, number>>;

const keyByNutrientId = new Map<number, NutrientKey>(
  Object.entries(USDA_NUTRIENT_IDS).map(([key, id]) => [id, key as NutrientKey]),
);

// SR Legacy quotes every field and has no line breaks inside fields, so a line-level parser
// is enough; anything unexpected throws instead of silently shifting columns.
export function parseCsvLine(line: string): string[] {
  const fields: string[] = [];
  let field = "";
  let inQuotes = false;
  for (let i = 0; i < line.length; i += 1) {
    const char = line[i];
    if (inQuotes) {
      if (char === '"' && line[i + 1] === '"') {
        field += '"';
        i += 1;
      } else if (char === '"') {
        inQuotes = false;
      } else {
        field += char;
      }
    } else if (char === '"') {
      inQuotes = true;
    } else if (char === ",") {
      fields.push(field);
      field = "";
    } else {
      field += char;
    }
  }
  if (inQuotes) throw new Error(`unterminated quote in CSV line: ${line.slice(0, 80)}`);
  fields.push(field);
  return fields;
}

export function collectUsdaNutrients(
  lines: readonly string[],
  fdcIds: ReadonlySet<number>,
): Map<number, UsdaNutrients> {
  const [headerLine, ...rows] = lines;
  const header = parseCsvLine(headerLine ?? "");
  const column = (name: string) => {
    const index = header.indexOf(name);
    if (index < 0) throw new Error(`food_nutrient.csv has no "${name}" column`);
    return index;
  };
  const fdcColumn = column("fdc_id");
  const nutrientColumn = column("nutrient_id");
  const amountColumn = column("amount");

  const result = new Map<number, UsdaNutrients>();
  rows.forEach((line, index) => {
    if (line.trim() === "") return;
    const fields = parseCsvLine(line);
    if (fields.length !== header.length) {
      throw new Error(`food_nutrient.csv line ${index + 2}: expected ${header.length} columns`);
    }
    const fdcId = Number(fields[fdcColumn]);
    const key = keyByNutrientId.get(Number(fields[nutrientColumn]));
    if (!fdcIds.has(fdcId) || !key) return;
    const amount = Number(fields[amountColumn]);
    if (!Number.isFinite(amount)) {
      throw new Error(`food_nutrient.csv line ${index + 2}: amount is not a number`);
    }
    result.set(fdcId, { ...result.get(fdcId), [key]: amount });
  });
  return result;
}

const scale = 10 ** NUTRIENT_STORAGE_DECIMALS;
const toStored = (value: number) => Math.round(value * scale) / scale;

function requireNutrient(key: string, nutrients: UsdaNutrients, name: NutrientKey): number {
  const value = nutrients[name];
  if (value === undefined) throw new Error(`${key}: USDA data has no ${name}`);
  return value;
}

export function buildSeedFoods(
  usda: readonly UsdaSource[],
  label: readonly LabelSource[],
  nutrients: ReadonlyMap<number, UsdaNutrients>,
): SeedFood[] {
  const fromUsda = usda.map((entry): SeedFood => {
    const key = `usda:${entry.fdcId}`;
    const data = nutrients.get(entry.fdcId);
    if (!data) throw new Error(`${key}: fdcId not found in food_nutrient.csv`);
    const fiber = data.fiber ?? null;
    return {
      key,
      name: entry.name,
      source: "usda",
      kcal100: toStored(requireNutrient(key, data, "kcal")),
      protein100: toStored(requireNutrient(key, data, "protein")),
      fat100: toStored(requireNutrient(key, data, "fat")),
      carbs100: toStored(
        calcNetCarbsPer100g(requireNutrient(key, data, "carbsByDifference"), fiber),
      ),
      fiber100: fiber === null ? null : toStored(fiber),
      sourceUrl: null,
      portions: entry.portions,
    };
  });
  const fromLabel = label.map(
    ({ slug, ...entry }): SeedFood => ({ key: `label:${slug}`, source: "label", ...entry }),
  );
  // Sorted output keeps git diffs of foods.json down to the rows that really changed.
  return [...fromUsda, ...fromLabel].sort((a, b) => a.key.localeCompare(b.key));
}
```

```ts
import { and, eq, isNotNull, isNull, notInArray, sql } from "drizzle-orm";
import { foodPortions, foods } from "@/db/schema";
import type { Db } from "@/db/types";
import type { SeedFood } from "./seed-schema";

// One transaction: a half-applied seed would leave foods without their portions.
export async function loadFoodSeed(
  db: Db,
  seed: readonly SeedFood[],
): Promise<{ upserted: number; retired: number }> {
  return db.transaction(async (tx) => {
    for (const { key, portions, ...values } of seed) {
      const [row] = await tx
        .insert(foods)
        .values({ ...values, seedKey: key, ownerId: null })
        .onConflictDoUpdate({
          target: foods.seedKey,
          // Clearing deleted_at revives a food whose key came back into the seed.
          set: { ...values, deletedAt: null, updatedAt: sql`now()` },
        })
        .returning({ id: foods.id });
      if (!row) throw new Error(`seed upsert returned no row for ${key}`);
      // Only shared portions are replaced; users' personal portions on this food stay.
      await tx
        .delete(foodPortions)
        .where(and(eq(foodPortions.foodId, row.id), isNull(foodPortions.userId)));
      if (portions.length > 0) {
        await tx
          .insert(foodPortions)
          .values(portions.map((portion) => ({ ...portion, foodId: row.id, userId: null })));
      }
    }

    const keys = seed.map((food) => food.key);
    // Soft delete: diary entries (stage 3) may still point at a food that left the seed.
    const retired = await tx
      .update(foods)
      .set({ deletedAt: sql`now()`, updatedAt: sql`now()` })
      .where(
        and(
          isNotNull(foods.seedKey),
          isNull(foods.deletedAt),
          keys.length > 0 ? notInArray(foods.seedKey, keys) : undefined,
        ),
      )
      .returning({ id: foods.id });
    return { upserted: seed.length, retired: retired.length };
  });
}
```

```tsx
"use client";

import Link from "next/link";
import { useRouter } from "next/navigation";
import { useEffect, useState } from "react";
import { FOOD_SEARCH } from "@/lib/config";
import type { FoodListItem } from "@/lib/foods/types";
import { ru } from "@/lib/i18n/ru";
import { FoodList } from "./food-list";

type SearchState =
  | { kind: "loading"; query: string }
  | { kind: "results"; query: string; items: FoodListItem[] }
  | { kind: "error"; query: string; message: string };

export function FoodSearch({ ownFoods }: { ownFoods: FoodListItem[] }) {
  const router = useRouter();
  const [input, setInput] = useState("");
  const [state, setState] = useState<SearchState | null>(null);
  const query = input.trim();

  useEffect(() => {
    if (!query) return;
    const controller = new AbortController();
    // Debounced and aborted on every keystroke, so a slow old response never replaces a newer one.
    const timer = setTimeout(async () => {
      setState({ kind: "loading", query });
      try {
        const response = await fetch(`/api/foods/search?q=${encodeURIComponent(query)}`, {
          signal: controller.signal,
        });
        if (response.status === 401) {
          router.push("/login");
          return;
        }
        if (!response.ok) {
          const message =
            response.status === 429 ? ru.foods.tooManyRequests : ru.foods.searchFailed;
          setState({ kind: "error", query, message });
          return;
        }
        const body = (await response.json()) as { items: FoodListItem[] };
        setState({ kind: "results", query, items: body.items });
      } catch {
        if (controller.signal.aborted) return;
        setState({ kind: "error", query, message: ru.common.networkError });
      }
    }, FOOD_SEARCH.debounceMs);
    return () => {
      clearTimeout(timer);
      controller.abort();
    };
  }, [query, router]);

  // A state for an older query is stale until the debounce for the current one fires.
  const current = state?.query === query ? state : null;

  return (
    <div className="flex flex-col gap-4">
      <label className="flex flex-col gap-1">
        <span className="sr-only">{ru.foods.searchLabel}</span>
        <input
          type="search"
          value={input}
          onChange={(event) => setInput(event.target.value)}
          placeholder={ru.foods.searchPlaceholder}
          maxLength={FOOD_SEARCH.maxQueryLength}
          autoComplete="off"
          enterKeyHint="search"
          className="min-h-11 rounded-lg border border-foreground/40 bg-background px-3 text-base"
        />
      </label>
      {!query ? (
        ownFoods.length > 0 ? (
          <section className="flex flex-col gap-2">
            <h2 className="text-sm font-medium text-foreground/60">{ru.foods.ownTitle}</h2>
            <FoodList items={ownFoods} />
          </section>
        ) : (
          <p className="text-foreground/60">{ru.foods.startTyping}</p>
        )
      ) : !current || current.kind === "loading" ? (
        <p className="text-foreground/60" aria-live="polite">
          {ru.foods.searching}
        </p>
      ) : current.kind === "error" ? (
        <p role="alert" className="text-red-600 dark:text-red-400">
          {current.message}
        </p>
      ) : current.items.length === 0 ? (
        <div className="flex flex-col gap-2">
          <p>{ru.foods.nothingFound}</p>
          <Link href="/foods/new" className="flex min-h-11 w-fit items-center underline">
            {ru.foods.addOwnInstead}
          </Link>
        </div>
      ) : (
        <FoodList items={current.items} />
      )}
    </div>
  );
}
```

```ts
export async function createFoodAction(_prev: FormState, formData: FormData): Promise<FormState> {
  const userId = await requireUserId();
  const parsed = foodSchema.safeParse(Object.fromEntries(formData));
  if (!parsed.success) return { fieldErrors: firstFieldErrors(parsed.error) };

  foodId: string,
  _prev: FormState,
  formData: FormData,
```

```tsx
      )}
      <Portions foodId={food.id} portions={portions} />
    </main>
  );
}
```
